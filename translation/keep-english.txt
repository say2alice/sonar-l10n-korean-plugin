# 번역하지 않고 영문 그대로 두는 값의 규칙입니다. 형식과 기준은 문서 저장소 번역 지침 §5에 있습니다.
# prefix <키 접두사>, pattern <값 전체와 일치하는 정규식>, key <키> 가운데 하나를 한 줄에 적습니다.

# 인자 이름을 빼면 영문자가 없는 값(인자와 기호만 있는 값, 빈 값)
pattern (\{[A-Za-z0-9_]+\}|[^A-Za-z])*

# 제품명과 상표만으로 된 값
pattern (Sonar|SonarSource|SonarQube|SonarQube for IDE|SonarQube Cloud|SonarQube Server|SonarQube Community Build|SonarLint|SonarCloud)
pattern (GitHub|GitHub Enterprise|GitLab|Bitbucket|Bitbucket Cloud|Bitbucket Server|Bitbucket Data Center|Azure DevOps|Azure DevOps Server|Jenkins|Jira|Slack)
# 제품명 뒤에 URL이나 ID가 붙은 입력 항목 이름(GitHub API URL, GitHub App ID)
pattern (GitHub|GitLab|Bitbucket Data Center|Azure DevOps)( API)? URL|(GitHub )?App ID(: \{[A-Za-z0-9_]+\})?
# 에디션 이름
pattern (Developer|Enterprise|Data Center) Edition

# 용어집에서 영문으로 두기로 한 용어만으로 된 값
pattern Code Smells?
pattern (Afferent|Efferent) [Cc]ouplings?

# 대문자 약어만으로 된 값(CSV, JSON, URL, PR)
pattern [A-Z]{2,5}

# 기능 이름으로 영문 그대로 두는 값
pattern AI Code Assurance|AI CODE ASSURANCE|AI CodeFix|Remediation Agent|Design & Architecture|Advanced Security|Advanced SAST( \(ASAST\))?|Compute Engine

# 보안과 규정 준수 표준의 이름만으로 된 값
pattern (OWASP (Top 10|Mobile Top 10|ASVS|MASVS)( for LLM)?|STIG ASD|PCI DSS|WCAG|EU CRA|CWE Top 25)( [0-9.]+| V[0-9R]+)?
pattern OWASP (Mobile )?Application Security Verification Standard|DISA Security Technical Implementation Guides

# 줄 번호 약식 표기(L12)
key issue.ncloc_x.short

# 지표 이름으로 영문 그대로 두는 값
pattern LCOM4

# 입력란의 예시 값(영문 슬러그)
key settings.almintegration.github.manifest.name.placeholder
key settings.almintegration.github.manifest.organization.placeholder
key settings.authentication.gitlab.form.allowedGroups.placeholder

# 프로그래밍 언어 이름만으로 된 설정 범주
key property.category.java

# 서비스 이름과 상태 인자만으로 된 알림(Sonar SCA {0})
key property.sca.message.updated

# 작은따옴표로 이스케이프한 꺾쇠 값. 번역하면 번역 품질 검사가 꺾쇠를 태그로 셉니다
key settings.default.no_value
key settings.default.complex_value
key settings.default.password

# Web API 문서의 이름과 HTTP 헤더 이름
pattern Web API( v2)?
key api_documentation.v2.request_subheader.request_content_type

# 용어집에서 영문으로 두기로 한 용어와 콜론만으로 된 값
key webhooks.delivery.payload

# 튜토리얼에서 가리키는 외부 제품의 메뉴, 권한, 범위 이름
key onboarding.create_project.pat_help.instructions.bitbucket_server.link
key onboarding.create_project.pat_help.gitlab.read_api_permission
